cmake_minimum_required(VERSION 3.21)
project(RuClaude LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network)
qt_standard_project_setup()
set(CMAKE_AUTOMOC ON)
qt_add_resources(ASSETS assets/resources.qrc)
add_library(ruclaude_common STATIC src/common.cpp src/common.h)
target_link_libraries(ruclaude_common PUBLIC Qt6::Core Qt6::Network)
if(WIN32)
  enable_language(RC)
  set(APP_ICON assets/icon.rc)
endif()
qt_add_executable(RuClaude WIN32 src/gui.cpp ${APP_ICON} ${ASSETS})
target_link_libraries(RuClaude PRIVATE ruclaude_common Qt6::Widgets)
qt_add_executable(RuClaudeService src/service.cpp src/firewall.cpp src/firewall.h)
target_link_libraries(RuClaudeService PRIVATE ruclaude_common)
if(WIN32)
  target_link_libraries(RuClaudeService PRIVATE advapi32)
endif()
install(TARGETS RuClaude RuClaudeService RUNTIME DESTINATION bin)
install(FILES assets/logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME ruclaude.png)
